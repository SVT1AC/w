  var sp=SPK[l.w];
  var ul=l.ul?'<ul>'+l.ul.map(function(x){return '<li>'+esc(x)+'</li>'}).join("")+'</ul>':'';
  return '<div class="say '+sp.side+'"><div class="who">'+avatar(l.w)+'<span>'+sp.name+'</span></div><div class="bub">'+(l.t?'<p>'+esc(l.t)+'</p>':'')+ul+'</div></div>'}

/* ---------- État ---------- */
var state;
function isIdentity(a){return a.every(function(v,i){return v===i+1})}
function init(){
  var ord;do{ord=shuffle([1,2,3,4,5,6])}while(isIdentity(ord));
  state={cur:0,warned:false,a:{},order:ord,ordTouched:false,opts:{},G:null};
  D.forEach(function(d){
    state.opts["d"+d.n+"k"]=shuffle(d.kwOpts).map(function(x){return {v:x,t:x}});
    state.opts["d"+d.n+"s"]=shuffle(STEPS).map(function(s){return {v:String(s.id),t:s.t}});
    d.extra.forEach(function(e){state.opts[e.id]=shuffle(e.opts).map(function(x){return {v:x,t:x}})});
  });
}
function qids(d){var l=["d"+d.n+"k","d"+d.n+"s"];d.extra.forEach(function(e){l.push(e.id)});return l}
function screenCount(i){
  if(i<6){var d=D[i];return {done:qids(d).filter(function(q){return state.a[q]}).length,total:qids(d).length}}
  return {done:state.ordTouched?1:0,total:1}}
function orderHits(){if(!state.ordTouched)return 0;var k=0;state.order.forEach(function(id,i){if(id===i+1)k++});return k}
function orderPts(){return Math.round(ORDER_PTS*orderHits()/6)}

/* ---------- Rendu ---------- */
function qRadio(id,num,text,cls){
  var o=state.opts[id].map(function(x){
    return '<label class="opt"><input type="radio" name="'+id+'" value="'+esc(x.v)+'" data-q="'+id+'"'+(state.a[id]===x.v?' checked':'')+'><span class="mark"></span><span>'+esc(x.t)+'</span></label>'}).join("");
  return '<fieldset class="q"><legend><span class="qn">'+num+'</span><span>'+esc(text)+'</span></legend><div class="opts '+cls+'">'+o+'</div></fieldset>'}

function rDisc(i){
  var d=D[i],pts=qids(d).length;
  var media=d.media==="jars"?jarsHTML():d.media==="table"?tableHTML():'';
  var qs=qRadio("d"+d.n+"k",1,"Quel est le mot clé de cette discussion ?","kw")+
         qRadio("d"+d.n+"s",2,"À quelle étape de la démarche correspond cette discussion ?","two");
  d.extra.forEach(function(e,j){qs+=qRadio(e.id,3+j,e.q,"")});
  return '<div class="phead"><span class="tag">Discussion '+d.n+' sur 6</span><span class="pts">'+pts+' point'+(pts>1?'s':'')+'</span></div>'+
    '<div class="scene" role="group" aria-label="Discussion '+d.n+'">'+d.lines.map(say).join("")+media+'</div>'+qs}

function rOrder(){
  var li=state.order.map(function(id,pos){
    var t=stepName(id);
    return '<li class="ocard"><span class="pos">'+(pos+1)+'</span><span class="otxt">'+esc(t)+'</span><span class="octl">'+
      '<button type="button" class="mv" data-act="mv" data-id="'+id+'" data-d="-1" aria-label="Monter : '+esc(t)+'"'+(pos===0?' disabled':'')+'>▲</button>'+
      '<button type="button" class="mv" data-act="mv" data-id="'+id+'" data-d="1" aria-label="Descendre : '+esc(t)+'"'+(pos===5?' disabled':'')+'>▼</button></span></li>'}).join("");
  return '<div class="phead"><span class="tag">Mise en ordre</span><span class="pts">'+ORDER_PTS+' points</span></div>'+
    '<h2>Remets les 6 étapes dans l\'ordre</h2>'+
    '<p class="consigne" style="margin-top:8px">Utilise les flèches pour placer les étapes de la démarche d\'investigation, de la première à la dernière.</p>'+
    '<ol class="olist" aria-label="Étapes à ordonner">'+li+'</ol>'+
    (state.ordTouched?'':'<p class="hint">Les étapes sont mélangées. Déplace-les pour les remettre en ordre.</p>')}

function refreshChrome(){
  var tot=0,done=0;
  for(var i=0;i<NSCREEN;i++){var c=screenCount(i);tot+=c.total;done+=c.done}
  $("#prog").textContent=done+" / "+tot+" réponses";
  $("#tabs").innerHTML=Array.apply(null,Array(NSCREEN)).map(function(_,i){
    var c=screenCount(i),dn=c.done===c.total;
    return '<button type="button" class="tab'+(dn?' done':'')+'" data-act="goto" data-i="'+i+'"'+(i===state.cur?' aria-current="step"':'')+' aria-label="'+(i<6?'Discussion '+(i+1):'Mise en ordre')+(dn?', terminée':'')+'"><b>'+(i+1)+'</b><span>'+(i<6?'Discussion':'Ordre')+'</span></button>'}).join("")}
function renderStage(){$("#stage").innerHTML=state.cur<6?rDisc(state.cur):rOrder()}
function renderNav(){
  var last=state.cur===NSCREEN-1,h="";
  h+=state.cur>0?'<button type="button" class="btn ghost" data-act="prev">Précédent</button>':'<span></span>';
  h+=last?'<button type="button" class="btn" data-act="finish">'+(state.warned?'Terminer quand même':'Terminer et voir ma note')+'</button>':'<button type="button" class="btn" data-act="next">Suivant</button>';
  $("#nav").innerHTML=h}
function renderQuiz(){refreshChrome();renderStage();renderNav();$("#warn").hidden=true}
function go(i){state.cur=Math.max(0,Math.min(NSCREEN-1,i));state.warned=false;renderQuiz();window.scrollTo(0,0)}

/* ---------- Notation ---------- */
function grade(){
  var secs=[];
  D.forEach(function(d){
    var items=[];
    var gk=state.a["d"+d.n+"k"],gs=state.a["d"+d.n+"s"];
    items.push({q:"Mot clé",ok:gk===d.kw,given:gk||"(pas de réponse)",exp:d.kw});
    items.push({q:"Étape de la démarche",ok:gs===String(d.step),given:gs?stepName(+gs):"(pas de réponse)",exp:stepName(d.step)});
    d.extra.forEach(function(e){var g=state.a[e.id];items.push({q:e.q,ok:g===e.ok,given:g||"(pas de réponse)",exp:e.ok})});
    secs.push({title:"Discussion "+d.n,items:items,score:items.filter(function(x){return x.ok}).length,max:items.length});
  });
  var k=orderHits();
  secs.push({title:"Mise en ordre des étapes",max:ORDER_PTS,score:orderPts(),
    items:[{q:"Ordre des 6 étapes ("+k+" sur 6 bien placées)",ok:k===6,
      given:state.ordTouched?state.order.map(function(id,i){return (i+1)+". "+stepName(id)}).join(" ; "):"(pas déplacé)",
      exp:STEPS.map(function(s){return s.id+". "+s.t}).join(" ; ")}]});
  var total=0;secs.forEach(function(s){total+=s.score});
  return {secs:secs,total:total}}
function verdict(n){
  if(n>=18)return ["Excellent travail","Tu connais très bien les étapes de la démarche d'investigation."];
  if(n>=14)return ["Très bien","Tu as compris l'essentiel. Regarde les réponses à revoir."];
  if(n>=10)return ["Bien, continue","Tu as compris le principe. Relis la leçon sur les 6 étapes."];
  return ["À revoir","Relis la leçon sur la démarche d'investigation, puis refais l'évaluation."]}
function reportText(){
  var n=($("#r-name")&&$("#r-name").value.trim())||"",c=($("#r-class")&&$("#r-class").value.trim())||"";
  var L=["Cahier de labo · La démarche d'investigation expérimentale (SVT 1AC)","Élève : "+(n||"…")+(c?" · Classe : "+c:""),"Note : "+state.G.total+" / 20",""];
  state.G.secs.forEach(function(s){L.push(s.title+" : "+s.score+" / "+s.max)});
  return L.join("\n")}

function showResult(){
  var G=grade();state.G=G;var v=verdict(G.total);
  var list=G.secs.map(function(s){
    var li=s.items.map(function(x){
      return '<li class="'+(x.ok?'ok':'ko')+'"><span class="badge">'+(x.ok?'Juste':'À revoir')+'</span><div><p>'+esc(x.q)+'</p>'+
        (x.ok?'':'<p class="exp">Ta réponse : '+esc(x.given)+'</p><p class="exp">Réponse attendue : '+esc(x.exp)+'</p>')+'</div></li>'}).join("");
    return '<details'+(s.score<s.max?' open':'')+'><summary><span>'+s.title+'</span><span>'+s.score+' / '+s.max+'</span></summary><ul class="items">'+li+'</ul></details>'}).join("");
  $("#result").innerHTML='<div class="sheet"><div class="res-top"><div class="grade"><span>'+G.total+'/20</span></div>'+
    '<div class="res-msg"><p class="eyebrow">Ta note</p><h2>'+v[0]+'</h2><p>'+v[1]+'</p></div></div>'+
    '<div class="mres">'+list+'</div>'+
    '<div class="send"><h2 style="font-size:1.25rem">Envoie ton compte rendu</h2><p>Écris ton nom et ta classe, puis copie le texte et envoie-le à ton enseignante.</p>'+
    '<div class="fields"><div class="field"><label class="lbl" for="r-name">Prénom et nom</label><input type="text" id="r-name" maxlength="60" autocomplete="off"></div>'+
    '<div class="field"><label class="lbl" for="r-class">Classe</label><input type="text" id="r-class" maxlength="20" autocomplete="off"></div></div>'+
    '<label class="sr" for="report">Compte rendu</label><textarea id="report" class="report" readonly></textarea>'+
    '<div><button type="button" class="btn" data-act="copy">Copier le compte rendu</button></div><p class="copied" id="copied" aria-live="polite"></p>'+
    '<div><button type="button" class="btn ghost" data-act="restart">Refaire l\'évaluation</button></div></div></div>';
  $("#report").value=reportText();
  $("#intro").hidden=true;$("#quiz").hidden=true;$("#result").hidden=false;window.scrollTo(0,0)}

/* ---------- Événements ---------- */
document.addEventListener("change",function(e){
  var t=e.target,q=t.dataset&&t.dataset.q;if(!q)return;
  state.a[q]=t.value;state.warned=false;$("#warn").hidden=true;renderNav();refreshChrome()});
document.addEventListener("input",function(e){
  if(e.target.id==="r-name"||e.target.id==="r-class")$("#report").value=reportText()});

document.addEventListener("click",function(e){
  var b=e.target.closest("[data-act]");if(!b)return;
  var act=b.dataset.act;
  if(act==="start"){$("#intro").hidden=true;$("#quiz").hidden=false;renderQuiz();window.scrollTo(0,0)}
  else if(act==="goto")go(+b.dataset.i);
  else if(act==="prev")go(state.cur-1);
  else if(act==="next")go(state.cur+1);
  else if(act==="mv"){
    var id=+b.dataset.id,d=+b.dataset.d,idx=state.order.indexOf(id),j=idx+d;
    if(j<0||j>5)return;
    var t=state.order[idx];state.order[idx]=state.order[j];state.order[j]=t;
    state.ordTouched=true;state.warned=false;$("#warn").hidden=true;
    renderStage();refreshChrome();renderNav();
    var nb=$('[data-act=mv][data-id="'+id+'"][data-d="'+d+'"]');
    if(nb&&nb.disabled)nb=$('[data-act=mv][data-id="'+id+'"][data-d="'+(-d)+'"]');
    if(nb)nb.focus({preventScroll:true})}
  else if(act==="finish"){
    var left=0;for(var i=0;i<NSCREEN;i++){var c=screenCount(i);left+=c.total-c.done}
    if(left>0&&!state.warned){
      state.warned=true;
      var w=$("#warn");w.textContent="Il reste "+left+(left>1?" réponses vides":" réponse vide")+". Reviens les remplir, ou termine quand même.";w.hidden=false;
      renderNav();return}
    showResult()}
  else if(act==="copy"){
    var ta=$("#report"),msg=$("#copied");
    var fb=function(){ta.focus();ta.select();msg.textContent="Sélectionne le texte et copie-le."};
    try{
      if(navigator.clipboard&&navigator.clipboard.writeText){navigator.clipboard.writeText(ta.value).then(function(){msg.textContent="Compte rendu copié."},fb)}
      else fb()
    }catch(err){fb()}}
  else if(act==="restart"){init();$("#result").hidden=true;$("#quiz").hidden=true;$("#intro").hidden=false;window.scrollTo(0,0)}
});

init();
$("#hero-scene").innerHTML=say({w:"S",t:"Par où commence-t-on une expérience ?"})+say({w:"A",t:"Suis-nous, on va le découvrir !"});
})();
